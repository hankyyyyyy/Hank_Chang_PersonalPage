function updateClock() {
  const now = new Date();
  // 取得時、分、秒，並使用 padStart(2, '0') 補齊兩位數
  const hh = String(now.getHours()).padStart(2, '0');
  const mm = String(now.getMinutes()).padStart(2, '0');
  const ss = String(now.getSeconds()).padStart(2, '0');

  // 更新 DOM 顯示：HH : MM : SS
  document.getElementById('clock-hours').textContent = hh;
  document.getElementById('clock-minutes').textContent = mm;
  document.getElementById('clock-seconds').textContent = ss;
}
// 每 1000 毫秒 (1 秒) 自動觸發一次
setInterval(updateClock, 1000);
